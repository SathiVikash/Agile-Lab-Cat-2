pipeline 
pipeline {
    agent any

    tools {
        maven 'Maven3'
        jdk 'JDK17'
    }

    environment {
        IMAGE_NAME = "studentapp"
        DOCKER_HUB = "yourdockerhub/studentapp"
    }

    stages {

        stage('Clone Code') {
            steps {
                git branch: 'main',
                url: 'https://github.com/username/studentapp.git'
            }
        }

        stage('Build') {
            steps {
                sh 'mvn clean compile'
            }
        }

        stage('Run Test') {
            steps {
                sh 'mvn test'
            }
        }

        stage('Publish JUnit Report') {
            steps {
                junit '**/target/surefire-reports/*.xml'
            }
        }

        stage('Store Report in GitHub') {
            steps {
                sh '''
                mkdir reports
                cp -r target/surefire-reports/* reports/
                git config user.email "jenkins@gmail.com"
                git config user.name "jenkins"
                git add reports/
                git commit -m "Added Test Reports"
                git push origin main
                '''
            }
        }

        stage('Build Docker Image') {
            steps {
                sh 'docker build -t $DOCKER_HUB .'
            }
        }

        stage('Push Docker Image') {
            steps {
                sh 'docker login -u user -p pass'
                sh 'docker push $DOCKER_HUB'
            }
        }

        stage('Deploy Kubernetes') {
            steps {
                sh 'kubectl apply -f k8s/'
            }
        }

    }
}
